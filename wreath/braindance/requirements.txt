# live3d runtime + engine-build deps. Intentionally UNPINNED — install latest and
# let uv resolve; pin only if a future combo actually breaks. Create the env with:
#
#   uv venv --python 3.12 .venv
#   uv pip install -r requirements.txt --torch-backend=auto
#
# --torch-backend=auto makes uv pick the CUDA wheel matching the local driver.

# --- depth model: inference + engine build ---
torch                 # DA2 inference (torch backend) + ONNX export + pre/post
transformers          # Depth Anything V2 from HuggingFace
tensorrt              # compile ONNX -> .engine, run it (fast path)
onnx                  # ONNX export/validation

# --- capture bridge + rendering ---
moderngl              # GL rendering (composite, mesh-warp DIBR, present)
glfw                  # override-redirect window + GL context on the glasses
PyOpenGL              # GL entrypoints moderngl/ctypes interop may need

# --- input + misc ---
python-xlib           # Super+Alt key chords, mouse grabs, pointer
Pillow                # HUD text rendering
numpy                 # arrays everywhere
